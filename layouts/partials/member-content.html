{{- /* Adapt only presentation. Fall back to source markup if any text is unaccounted for. */ -}} {{- $content := . -}} {{- $parts := split $content `
` -}} {{- if eq (len $parts) 4 -}} {{- $prefix := index $parts 0 -}} {{- $galleries := findRE `
[\s\S]*?
\s*
` $prefix -}} {{- $labels := findRE `]*>[\s\S]*?` $prefix -}} {{- $valid := and (eq (len $galleries) 1) (eq (len $labels) 1) -}} {{- $groups := "" -}} {{- range after 1 $parts -}} {{- $headings := findRE `]*>[\s\S]*?` . -}} {{- $members := findRE `]*>[\s\S]*?` . -}} {{- if and (eq (len $headings) 1) (gt (len $members) 0) -}} {{- $heading := index $headings 0 -}} {{- $entries := delimit $members "" -}} {{- $oldText := replaceRE `\s+` "" (plainify .) -}} {{- $newText := replaceRE `\s+` "" (plainify (printf "%s%s" $heading $entries)) -}} {{- if ne $oldText $newText -}}{{- $valid = false -}}{{- end -}} {{- $heading = replace (replace $heading "" "" -}} {{- $groups = printf "%s
%s
    %s
" $groups $heading $entries -}} {{- else -}}{{- $valid = false -}}{{- end -}} {{- end -}} {{- if $valid -}} {{- $gallery := index $galleries 0 -}} {{- $label := index $labels 0 -}} {{- if eq (replaceRE `\s+` "" (plainify $prefix)) (replaceRE `\s+` "" (plainify (printf "%s%s" $gallery $label))) -}} {{- $label = replace (replace $label "" "" -}} {{- /* Render gallery navigation up front so initialization cannot move the roster/footer. */ -}} {{- $dots := "" -}} {{- $thumbnails := "" -}} {{- $images := findRE `]+>` $gallery -}} {{- $firstCaption := "" -}} {{- range $index, $image := $images -}} {{- $alt := replaceRE `^.*?\salt="([^"]*)".*$` "$1" $image -}} {{- if eq $index 0 -}}{{- $firstCaption = htmlEscape $alt -}}{{- end -}} {{- $slideLabel := htmlEscape (printf "%s %d: %s" (i18n "slideLabel") (add $index 1) $alt) -}} {{- $active := eq $index 0 -}} {{- $dotClass := "dot" -}} {{- if $active -}}{{- $dotClass = "dot active" -}}{{- end -}} {{- $dots = printf `%s` $dots $dotClass $slideLabel $active -}} {{- $thumbnail := replaceRE `\salt="[^"]*"` ` alt=""` $image -}} {{- $thumbnails = printf `%s` $thumbnails $slideLabel $active $thumbnail -}} {{- end -}} {{- $gallery = replace $gallery `
` (printf `
%s
` $dots) -}} {{- $meta := printf `
%s01 / %02d
` $firstCaption (len $images) -}} {{- $content = printf `
%s%s
%s
%s
%s
` $gallery $meta $thumbnails $label $groups -}} {{- end -}} {{- end -}} {{- end -}} {{- return $content -}}
